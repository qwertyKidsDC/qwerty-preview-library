Services → DismantleTargeting · COMPLETE NEW MODULE
local Players = game:GetService("Players")

local DismantleTargeting = {}

function DismantleTargeting.Find(attacker, originRoot, range, halfAngle)
    local bestPlayer = nil
    local bestDistance = math.huge

    for _, candidate in Players:GetPlayers() do
        if candidate ~= attacker then
            local character = candidate.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")

            if humanoid and humanoid.Health > 0 and root then
                local offset = root.Position - originRoot.Position
                local distance = offset.Magnitude

                if distance > 0 and distance <= range and distance < bestDistance then
                    local forward = originRoot.CFrame.LookVector
                    local dot = forward:Dot(offset.Unit)

                    if dot >= math.cos(math.rad(halfAngle)) then
                        local params = RaycastParams.new()
                        params.FilterType = Enum.RaycastFilterType.Exclude
                        params.FilterDescendantsInstances = {attacker.Character}

                        local hit = workspace:Raycast(originRoot.Position, offset, params)
                        if hit and hit.Instance:IsDescendantOf(character) then
                            bestPlayer = candidate
                            bestDistance = distance
                        end
                    end
                end
            end
        end
    end

    return bestPlayer
end

return DismantleTargeting
